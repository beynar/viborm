/** D-40 PROBE — PostgreSQL. Removed once its receipt is taken. */
import { convertBigIntToNumber } from "@adapters/shared/result-parsing";
import { PgDriver } from "@drivers/pg";
import { describe, it } from "vitest";
import { runLegProbe } from "./o2-leg-probe";
import { TEST_CONNECTION_STRING } from "./pg-fixtures";

const describeIf = TEST_CONNECTION_STRING ? describe : describe.skip;

describeIf("D-40 probe — the PostgreSQL adapter parseResult leg", () => {
  it("records every raw the bigint leg decides about", async () => {
    await runLegProbe({
      provider: "pg",
      driver: new PgDriver({ databaseUrl: TEST_CONNECTION_STRING }),
      legDecision: (raw) => convertBigIntToNumber(raw),
    });
  }, 120_000);
});
