#!/bin/zsh
# The whole extended-local estate, the provider stage, the modes, core, typecheck and the floors — in the N1 worktree.
set -u
cd /private/tmp/viborm-n1 || exit 2
S=/private/tmp/claude-501/-Users-arnaud-code-viborm/c2c775da-2927-4590-8677-3bb0f5d1aa98/scratchpad
O=$S/n1/estate; mkdir -p "$O"; : > "$O/RESULTS.txt"
export TMPDIR=/private/tmp/viborm-n1-estate-tmp; mkdir -p "$TMPDIR"
stage() { local name=$1; shift; "$@" > "$O/$name.log" 2>&1; local code=$?; echo "$name exit=$code $(grep -E '^\s*(Test Files|Tests) ' "$O/$name.log" | tail -2 | tr -s ' ' | tr '\n' ';')" >> "$O/RESULTS.txt"; }
stage fixed pnpm test:all --only "Raptor 3 fixed"
for n in $(seq 1 13); do stage "ordinary-$n" pnpm test:all --only "extended-local shard $n/"; done
for n in $(seq 1 8); do stage "shard-$n" pnpm test:all --only "shared-family shard $n/"; done
for n in 1 2; do stage "pglite-$n" pnpm test:all --only "imported-pglite shard $n/"; done
stage provider pnpm test:all --only "raptor3-provider"
for m in g2-baseline g2-contracts g3-transaction-array; do stage "mode-$m" node scripts/run-raptor3.mjs "$m"; done
stage typecheck node scripts/run-typecheck.mjs
stage core pnpm test:core
pnpm test:coverage:query-engine-core > "$O/coverage-query-engine-core.log" 2>&1; echo "coverage:query-engine-core exit=$? $(grep -E 'All files' "$O/coverage-query-engine-core.log" | head -1 | tr -s ' ')" >> "$O/RESULTS.txt"
pnpm test:coverage:policy > "$O/coverage-policy.log" 2>&1; echo "coverage:policy exit=$?" >> "$O/RESULTS.txt"
echo DONE >> "$O/RESULTS.txt"
